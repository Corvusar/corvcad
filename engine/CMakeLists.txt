cmake_minimum_required(VERSION 3.21)
project(CorvCADEngine VERSION 0.1.0 LANGUAGES CXX)
option(CORVCAD_WITH_OCCT "Enable Open CASCADE STEP export" OFF)
add_library(corvcad_engine STATIC src/engine.cpp src/net_flow.cpp src/eda.cpp src/authoring.cpp src/schematic_route.cpp src/project.cpp src/library.cpp src/gerber.cpp src/calculator.cpp src/c_api.cpp src/schematic_session.cpp src/schematic_session_c_api.cpp src/board_native_c_api.cpp)
target_compile_features(corvcad_engine PUBLIC cxx_std_20)
target_include_directories(corvcad_engine PUBLIC include)
target_sources(corvcad_engine PRIVATE src/mechanical_session_c_api.cpp)
if(CORVCAD_WITH_OCCT)
  find_package(OpenCASCADE REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms DataExchange)
  target_compile_definitions(corvcad_engine PRIVATE CORVCAD_WITH_OCCT=1)
  target_include_directories(corvcad_engine PRIVATE ${OpenCASCADE_INCLUDE_DIR})
  target_link_libraries(corvcad_engine PUBLIC ${OpenCASCADE_LIBRARIES})
endif()
add_executable(corvcad_cli src/main.cpp)
target_link_libraries(corvcad_cli PRIVATE corvcad_engine)
add_executable(corvcad_eda_cli src/eda_main.cpp)
target_link_libraries(corvcad_eda_cli PRIVATE corvcad_engine)
add_executable(corvcad_schematic_session_cli src/schematic_session_main.cpp)
target_link_libraries(corvcad_schematic_session_cli PRIVATE corvcad_engine)
enable_testing()
add_executable(corvcad_mechanical_boundary_tests tests/test_mechanical_boundary.cpp)
target_link_libraries(corvcad_mechanical_boundary_tests PRIVATE corvcad_engine)
add_test(NAME mechanical_boundary COMMAND corvcad_mechanical_boundary_tests)
add_executable(corvcad_tests tests/test_engine.cpp)
target_link_libraries(corvcad_tests PRIVATE corvcad_engine)
target_compile_definitions(corvcad_tests PRIVATE CORVCAD_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/demo.kicad_pcb")
add_test(NAME engine COMMAND corvcad_tests)
add_executable(corvcad_eda_tests tests/test_eda.cpp)
target_link_libraries(corvcad_eda_tests PRIVATE corvcad_engine)
target_compile_definitions(corvcad_eda_tests PRIVATE CORVCAD_ROUNDTRIP="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/roundtrip/RoundTrip.kicad_sch")
add_test(NAME eda COMMAND corvcad_eda_tests)
add_executable(corvcad_project_tests tests/test_project.cpp)
target_link_libraries(corvcad_project_tests PRIVATE corvcad_engine)
add_test(NAME project COMMAND corvcad_project_tests)
add_executable(corvcad_authoring_tests tests/test_authoring.cpp)
target_link_libraries(corvcad_authoring_tests PRIVATE corvcad_engine)
target_compile_definitions(corvcad_authoring_tests PRIVATE CORVCAD_ROUNDTRIP_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/roundtrip")
add_test(NAME authoring COMMAND corvcad_authoring_tests)
add_executable(corvcad_library_tests tests/test_library.cpp)
target_link_libraries(corvcad_library_tests PRIVATE corvcad_engine)
add_test(NAME library COMMAND corvcad_library_tests)
add_executable(corvcad_tools_tests tests/test_tools.cpp)
target_link_libraries(corvcad_tools_tests PRIVATE corvcad_engine)
target_compile_definitions(corvcad_tools_tests PRIVATE CORVCAD_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
add_test(NAME tools COMMAND corvcad_tools_tests)
add_executable(corvcad_schematic_session_tests tests/test_schematic_session.cpp)
target_link_libraries(corvcad_schematic_session_tests PRIVATE corvcad_engine)
add_test(NAME schematic_session COMMAND corvcad_schematic_session_tests)
